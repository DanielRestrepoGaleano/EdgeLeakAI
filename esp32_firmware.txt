#include <Arduino.h>
#include <WiFi.h>
#include <HTTPClient.h>

// =================================================
// SCRIPT FINAL v3.0: EdgeLeak AI (Telemetry Node con Detección de Patrones)
// =================================================

// --- Configuración de Red y API ---
const char* ssid     = "DANIEL";          // CAMBIAR
const char* password = "43723298";        // CAMBIAR
const char* apiUrl   = "http://192.168.1.2:8080/api/sensor"; 
const char* apiKey   = "edgeleak-share-2024";

// --- Mapeo de Pines (ESP32 de 30 pines) ---
const int PIN_SENSOR_FLUJO = 32;   
const int PIN_RUIDO_ANALOG = 34;   
const int PIN_RUIDO_DIGITAL = 35;  

// --- Constantes de Calibración ---
const float FACTOR_CALIBRACION = 450.0; 
const int INTERVALO_LECTURA = 5000; // 5 segundos

// --- Variables Globales de Flujo ---
volatile int contadorPulsos = 0; 
float caudalLitrosPorMinuto = 0.0; 
unsigned long tiempoAnterior = 0;  

// --- Variables Globales de Patrón de Ruido (Goteo) ---
float promedioSilencio = 1100.0; // Inicia en el piso de ruido calibrado
int picosDetectados = 0;
unsigned long ultimaGota = 0;
const int UMBRAL_SENSIBILIDAD = 150; // Diferencia para considerar un "pico"
const unsigned long DEBOUNCE_GOTEO = 200; // Evita contar el mismo ruido múltiple veces (ms)

// --- Funciones de Interrupción ---
void IRAM_ATTR contarPulso() {
  contadorPulsos++;
}

// --- Función para auto-calibrar el silencio y detectar goteo en tiempo real ---
void escucharPatrones() {
  int lecturaActual = analogRead(PIN_RUIDO_ANALOG);
  int limiteActivacion = (int)promedioSilencio + UMBRAL_SENSIBILIDAD;

  if (lecturaActual > limiteActivacion) {
    unsigned long ahora = millis();
    // Validar si es un pico nuevo usando debounce
    if (ahora - ultimaGota > DEBOUNCE_GOTEO) {
      picosDetectados++;
      ultimaGota = ahora;
    }
  } else {
    // Si hay silencio, actualizamos dinámicamente la línea base (Promedio Móvil Exponencial)
    // Esto permite que el sistema se adapte lentamente si el ruido ambiente cambia en la noche
    promedioSilencio = (promedioSilencio * 0.999) + (lecturaActual * 0.001);
  }
}

// --- Función para mantener conexión robusta ---
void verificarWiFi() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Reconectando Wi-Fi...");
    WiFi.disconnect();
    WiFi.reconnect();
    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 5000) {
      delay(500);
      Serial.print(".");
    }
    Serial.println();
  }
}

// --- Cálculo de Caudal ---
void calcularCaudal() {
  noInterrupts(); 
  int pulsos = contadorPulsos;
  contadorPulsos = 0; 
  interrupts(); 
  caudalLitrosPorMinuto = (float)pulsos / FACTOR_CALIBRACION * (60000.0 / INTERVALO_LECTURA);
}

// --- Envío HTTP ---
void mostrarYEnviarDatos() {
  int nivelRuidoActual = analogRead(PIN_RUIDO_ANALOG);
  int golpeDetectado = digitalRead(PIN_RUIDO_DIGITAL);
  
  // Guardamos los picos procesados en estos 5 segundos y reiniciamos el contador local
  int picosParaEnviar = picosDetectados;
  picosDetectados = 0;

  Serial.print("Flujo: ");
  Serial.print(caudalLitrosPorMinuto, 2);
  Serial.print(" | Silencio Base: ");
  Serial.print((int)promedioSilencio);
  Serial.print(" | Ruido Inst: ");
  Serial.print(nivelRuidoActual);
  Serial.print(" | Picos(5s): ");
  Serial.println(picosParaEnviar);

  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(apiUrl);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("x-api-key", apiKey);

    // NUEVO JSON CON LA VARIABLE 'picos'
    String body = "{\"ruido\":" + String(nivelRuidoActual) + 
                  ",\"flujo\":" + String(caudalLitrosPorMinuto, 2) + 
                  ",\"picos\":" + String(picosParaEnviar) + "}";
    
    int httpResponseCode = http.POST(body);
    if (httpResponseCode <= 0) {
      Serial.print("Error POST: ");
      Serial.println(http.errorToString(httpResponseCode).c_str());
    }
    http.end(); 
  }
}

void setup() {
  Serial.begin(115200);
  
  pinMode(PIN_SENSOR_FLUJO, INPUT_PULLUP);
  pinMode(PIN_RUIDO_ANALOG, INPUT);
  pinMode(PIN_RUIDO_DIGITAL, INPUT);
  attachInterrupt(digitalPinToInterrupt(PIN_SENSOR_FLUJO), contarPulso, FALLING);

  Serial.println("\n=========================================");
  Serial.println("  Iniciando EdgeLeak AI v3.0 (Pattern AI)");
  Serial.println("=========================================\n");
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nIP Asignada: ");
  Serial.println(WiFi.localIP());
}

void loop() {
  verificarWiFi();
  escucharPatrones(); // El procesador escucha la acústica miles de veces por segundo
  
  if (millis() - tiempoAnterior > INTERVALO_LECTURA) {
    tiempoAnterior = millis(); 
    calcularCaudal();
    mostrarYEnviarDatos();
  }
}
